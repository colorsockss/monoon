#!/bin/sh
# Adds this game to the desktop menu, pointing at this folder.
set -eu
here=$(CDPATH= cd -- "$(dirname -- "$0")" && pwd)
target="${XDG_DATA_HOME:-$HOME/.local/share}/applications"
mkdir -p "$target"
sed -e "s|^Exec=.*|Exec=\"$here/MoNoonPlayer\"|" -e "s|^Icon=.*|Icon=$here/KeyRoom.png|" "$here/KeyRoom.desktop" > "$target/monoon-64fb4cd624695cecba6d9a49e61f42e7.desktop"
echo "Installed $target/monoon-64fb4cd624695cecba6d9a49e61f42e7.desktop"
