using System.Diagnostics;
using Brutal.SdlApi;
using Engine.Core;
using Engine.Desktop;
string root = "/Users/paare/codex_project/brutal_engine/.worktrees/phase1/samples/KeyRoom/Assets/Fonts/";
string[] labels = ["가방", "휴대용 나침반 / 보관 중", "의뢰 완료 / 황동 열쇠", "나침반 관리", "닫기", "English", "English [L]", "가방 [I]", "조작법",
    "문이 열렸다. 동쪽으로 가자.", "Quest complete / Brass key", "Pocket compass / In bag"];
unsafe
{
    var surface = Sdl.PInvoke.CreateSurface(800, 450, SdlPixelFormat.Rgba32);
    var renderer = Sdl.Render.CreateSoftwareRenderer(surface);
    using var canvas = new SdlCanvas(renderer);
    ICanvas2D c = canvas;
    c.UseFonts([root + "GowunDodum-Regular.ttf", root + "NanumGothic-Regular.ttf"]);
    void Frame()
    {
        float y = 10;
        foreach (var label in labels) { float w = c.MeasureText(label, 19); c.DrawText(label, 20 + w * 0, y, 19, System.Drawing.Color.White); y += 30; }
    }
    var sw = Stopwatch.StartNew();
    Frame();
    double cold = sw.Elapsed.TotalMilliseconds;
    for (int i = 0; i < 200; i++) Frame();
    var times = new List<double>();
    long bytes = 0;
    for (int run = 0; run < 7; run++)
    {
        long b0 = GC.GetAllocatedBytesForCurrentThread();
        sw.Restart();
        for (int i = 0; i < 1000; i++) Frame();
        times.Add(sw.Elapsed.TotalMilliseconds);
        bytes = GC.GetAllocatedBytesForCurrentThread() - b0;
    }
    times.Sort();
    Console.WriteLine($"cold first frame (12 labels rasterized): {cold:F2} ms");
    Console.WriteLine($"cached frame (12 labels measure+draw): median {times[3]:F1} us, range {times[0]:F1}-{times[^1]:F1} us, {bytes / 1000.0:F1} bytes/frame");
}
